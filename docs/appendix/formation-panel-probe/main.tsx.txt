type Box = { l: number; r: number; t: number; b: number };
const box = (el: Element): Box => {
  const r = el.getBoundingClientRect();
  return { l: r.left, r: r.right, t: r.top, b: r.bottom };
};
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));

async function measure() {
  await sleep(1500);
  const tab = [...document.querySelectorAll('[role="tab"]')].find((t) =>
    t.textContent?.includes("Substitutions"),
  ) as HTMLElement | undefined;
  tab?.click();
  await sleep(500);
  const pitchSection = document.querySelector('section[aria-label^="Tactical pitch"]');
  const tokens = [...(pitchSection?.querySelectorAll('[role="button"]') ?? [])] as HTMLElement[];
  const footprints = tokens.map((el) => ({ el, b: box((el.firstElementChild as Element) ?? el) }));
  let overlaps = 0;
  const pairs: string[] = [];
  for (let i = 0; i < footprints.length; i++)
    for (let j = i + 1; j < footprints.length; j++) {
      const a = footprints[i].b,
        b = footprints[j].b;
      const ox = Math.min(a.r, b.r) - Math.max(a.l, b.l),
        oy = Math.min(a.b, b.b) - Math.max(a.t, b.t);
      if (ox > 1 && oy > 1) {
        overlaps++;
        pairs.push(
          `${tokens[i].getAttribute("aria-label")}/${tokens[j].getAttribute("aria-label")}:${Math.round(ox)}x${Math.round(oy)}`,
        );
      }
    }
  const pitch = pitchSection ? box(pitchSection.firstElementChild as Element) : null;
  const outside = pitch
    ? footprints.filter(
        ({ b }) => b.l < pitch.l - 1 || b.r > pitch.r + 1 || b.t < pitch.t - 1 || b.b > pitch.b + 1,
      ).length
    : -1;
  // text that is cut off: a name or a role line wider than its box
  const clipped = [...(pitchSection?.querySelectorAll("span") ?? [])]
    .filter(
      (s) =>
        !s.classList.contains("sr-only") &&
        s.scrollWidth > s.clientWidth + 1 &&
        getComputedStyle(s).overflow !== "visible",
    )
    .map((s) => s.textContent);
  // select a player and walk the panel's controls
  tokens[5]?.click();
  await sleep(200);
  const panel = pitchSection?.parentElement as HTMLElement | null;
  const controls = [
    ...(panel?.querySelectorAll("button, [role=combobox], [role=button]") ?? []),
  ] as HTMLElement[];
  const vw = document.documentElement.clientWidth,
    vh = document.documentElement.clientHeight;
  const covered: string[] = [];
  const offscreenX: string[] = [];
  for (const c of controls) {
    const r = c.getBoundingClientRect();
    if (r.width === 0 || r.height === 0) continue;
    if (r.right > vw + 1 || r.left < -1) offscreenX.push(c.textContent?.trim().slice(0, 24) ?? "?");
    c.scrollIntoView({ block: "center" });
    const rr = c.getBoundingClientRect();
    const hit = document.elementFromPoint(rr.left + rr.width / 2, rr.top + rr.height / 2);
    if (hit && hit !== c && !c.contains(hit))
      covered.push(c.getAttribute("aria-label") ?? c.textContent?.trim().slice(0, 24) ?? "?");
  }
  // focus ring: a token focused by keyboard shows a ring (box-shadow) in the real stylesheet
  const probeToken = tokens[0];
  probeToken?.focus({ focusVisible: true } as FocusOptions);
  const ring = probeToken ? getComputedStyle(probeToken).boxShadow : "none";
  const docOverflowX = document.documentElement.scrollWidth - document.documentElement.clientWidth;
  const wide = [...document.querySelectorAll("body *")]
    .map((e) => ({ e, r: e.getBoundingClientRect().right }))
    .filter((x) => x.r > vw + 1)
    .sort((a, b) => b.r - a.r)
    .slice(0, 4)
    .map(
      (x) =>
        `${x.e.tagName}.${(x.e.getAttribute("class") ?? "").split(" ").slice(0, 3).join(".")}:${Math.round(x.r)}`,
    );
  const panelBox = panel ? box(panel) : null;
  const heading =
    [...document.querySelectorAll("h3")].find(
      (h) =>
        h.textContent?.startsWith("#") || h.closest("section")?.id === "formation-panel-selection",
    )?.textContent ?? null;
  const result = {
    formation,
    variant,
    longNames,
    viewport: [vw, vh],
    tokens: tokens.length,
    overlappingPairs: overlaps,
    pairs,
    outsidePitch: outside,
    clippedTextSpans: clipped,
    pitch: pitch ? [Math.round(pitch.r - pitch.l), Math.round(pitch.b - pitch.t)] : null,
    controls: controls.length,
    offscreenX,
    covered,
    focusRing: ring !== "none",
    documentOverflowX: docOverflowX,
    selectedHeading: heading,
    wideElements: wide,
    bodyScrollW: document.body.scrollWidth,
    panelBox: panelBox
      ? [
          Math.round(panelBox.l),
          Math.round(panelBox.r),
          Math.round(panelBox.t),
          Math.round(panelBox.b),
        ]
      : null,
  };
  const pre = document.getElementById("probe-result");
  if (pre) pre.textContent = JSON.stringify(result);
  window.scrollTo(0, 0);
  tab?.click();
  tab?.click();
}

await i18nReady;

void measure();
